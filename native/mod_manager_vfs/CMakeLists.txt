cmake_minimum_required(VERSION 3.24)
project(ModBoxVirtualFilespace LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ModBox virtual filespace currently requires x64 Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(modbox_minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(modbox_minhook PUBLIC vendor/minhook/include PRIVATE vendor/minhook/src vendor/minhook/src/hde)
add_library(modbox_vfs SHARED filespace.cpp injection.cpp)
option(MODBOX_VFS_TESTING "Build fixture-only fault injection controls" OFF)
if(MODBOX_VFS_TESTING)
  target_compile_definitions(modbox_vfs PRIVATE MODBOX_VFS_TESTING)
endif()
target_compile_definitions(modbox_vfs PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(modbox_vfs PRIVATE /W4 /EHsc /guard:cf)
target_link_options(modbox_vfs PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(modbox_vfs PRIVATE modbox_minhook bcrypt user32)
add_executable(modbox_vfs_fixture fixture.cpp)
target_compile_definitions(modbox_vfs_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(modbox_vfs_fixture PRIVATE /W4 /EHsc)
add_executable(modbox_window_fixture window_fixture.cpp)
target_link_libraries(modbox_window_fixture PRIVATE user32)
target_compile_definitions(modbox_window_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
